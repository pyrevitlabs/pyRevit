    _events

    Retain CPython callbacks on native WPF objects for the lifetime of a form.

    Every callback a form attaches is recorded in the form's _EventRegistry so release detaches all of them together. A callback that raises behaves as under IronPython whenever Python code is waiting below it: the error leaves the event and reaches that code, for example the caller of RaiseEvent or ShowDialog. When no Python code is below it, as for input to a modeless window or a dockable page after its command has finished, the error is logged instead of reaching the WPF dispatcher, where it could terminate Revit.

    def hook_template_events(registry, key, handlers):
    """Attach ``handlers`` directly to each element instance marked with ``key``.

    ``key`` is a ``PythonEvents.Hook`` value from ``_xaml.HookedEvent``; WPF
    hands over every instance it creates, including instances created while
    the markup is parsed, so call this before loading it. Each handler is
    guarded like any other form callback, and ``release`` stops delivery and
    detaches them.

    An instance is tracked from creation until it unloads: it then drops its
    handlers and is forgotten, and gets them back if it loads again. Its
    ``Loaded``/``Unloaded`` subscriptions stay, and are removed from every
    instance still alive by ``release``, which reaches them through weak
    references. An instance that never loads stays tracked until ``release``.

    Args:
        registry (_EventRegistry): The form's registry.
        key (str): The element's hook key.
        handlers (list[tuple[str, callable]]): (event name, handler) pairs.

    Returns:
        callable: Stops delivery without detaching handlers already attached,
            for a load that failed before the form owns the markup.
    """
    callbacks = [(name, registry.guard(handler)) for name, handler in handlers]
    attached = {}
    observed = []
    state = {"released": False, "prune_at": 64}

    def find(element):
        for entry in attached.get(_identity(element), ()):
            if System.Object.ReferenceEquals(element, entry[0]):
                return entry
        return None

    def bind(element):
        if state["released"] or find(element) is not None:
            return
        bound = []
        try:
            for event_name, callback in callbacks:
                event = getattr(element, event_name)
                event += callback
                bound.append((event_name, callback))
        except Exception as error:
            _report(error)
        attached.setdefault(_identity(element), []).append((element, bound))

    def unbind(entry):
        element, bound = entry
        for event_name, callback in reversed(bound):
            event = getattr(element, event_name)
            event -= callback

    def forget(sender, args):
        entry = find(sender)
        if entry is None:
            return
        unbind(entry)
        key = _identity(sender)
        bucket = [known for known in attached[key] if known is not entry]
        if bucket:
            attached[key] = bucket
        else:
            del attached[key]

    def reload(sender, args):
        bind(sender)

    loaded = Windows.RoutedEventHandler(reload)
    unloaded = Windows.RoutedEventHandler(forget)

    def is_loadable(element):
        return isinstance(
            element, (Windows.FrameworkElement, Windows.FrameworkContentElement)
        )

    def observe(element):
        if len(observed) >= state["prune_at"]:
            observed[:] = [reference for reference in observed if reference.IsAlive]
            state["prune_at"] = max(64, 2 * len(observed))
        observed.append(System.WeakReference(element))

    def stop_observing(element):
        element.Loaded -= loaded
        element.Unloaded -= unloaded

    def attach(element):
        bind(element)
        if is_loadable(element):
            try:
                element.Loaded += loaded
                element.Unloaded += unloaded
                observe(element)
            except Exception as error:
                _report(error)

    def detach():
        PythonEvents.Detach(key)

    def release():
        state["released"] = True
        for bucket in list(attached.values()):
            for entry in bucket:
                unbind(entry)
        attached.clear()
        for reference in observed:
            element = reference.Target
            if element is not None:
                _attempt(stop_observing, element)
        del observed[:]

    PythonEvents.Attach(key, attach)
    registry.on_release(detach)
    registry.on_release(release)
    return detach

    def replay_initial_state(control, event_name, callback):
    """Run ``callback`` once for state the XAML parser set before it attached.

    Handlers are attached after ``XamlReader.Load``, so ``Checked`` for a
    toggle that loads checked and ``SelectionChanged`` for a selector that
    loads with a selection were raised before any handler existed. The loader
    calls this only for events whose ``NamedEvent.replay`` is set, which is
    where IronPython's attribute-ordered loader would have delivered them;
    the loaded state then decides whether anything is raised. ``TextChanged``,
    ``ValueChanged`` and other events raised during parsing are not replayed.
    """
    primitives = framework.Controls.Primitives
    if event_name == "Checked":
        if isinstance(control, primitives.ToggleButton) and control.IsChecked is True:
            args = Windows.RoutedEventArgs(
                primitives.ToggleButton.CheckedEvent, control
            )
            callback(control, args)
    elif event_name == "SelectionChanged":
        if isinstance(control, primitives.Selector) and control.SelectedIndex >= 0:
            selected = getattr(control, "SelectedItems", None)
            added = System.Collections.ArrayList()
            for item in selected if selected is not None else [control.SelectedItem]:
                added.Add(item)
            args = framework.Controls.SelectionChangedEventArgs(
                primitives.Selector.SelectionChangedEvent,
                System.Collections.ArrayList(),
                added,
            )
            args.Source = control
            callback(control, args)

    def is_event_attribute(tag, attribute):
    """Return whether a XAML attribute on ``tag`` names a CLR event of that type.

    ``tag`` is a WPF type name or ``{clr-namespace:Ns;assembly=Asm}Name``;
    custom types resolve through the loaded assemblies. Unresolvable types
    report False, leaving ``_xaml._EVENTS`` as the fallback.
    """
    key = (tag, attribute)
    if key not in _event_attributes:
        owner = _clr_type(tag)
        _event_attributes[key] = (
            owner is not None and owner.GetEvent(attribute) is not None
        )
    return _event_attributes[key]

    bind_template_event(registry, root, tag, uid, event_name, handler)

    Route an event declared inside a template to handler.

    Template instances do not exist when the XAML loads, so their elements are matched by Uid (and by type, for WPF controls). Bubbling events with a preview counterpart get a direct handler on each matching element the first time the tunnelling preview event reaches it, which preserves WPF's handler order and Handled semantics. Other routed events are matched from a handler on root. Direct mouse-button events are rebuilt from MouseDown/MouseUp filtered by button and click count; double clicks accept the left or right button with ClickCount == 2, as Control.MouseDoubleClick does, and a double click handler also runs after the press was marked handled, as that control's own event does. An element that unloads drops its direct handler and is rediscovered if it loads again. tag is the element's WPF type name, or {namespace}Name for a type outside the WPF namespace, which is then matched by Uid alone.

    Note

    An event with no preview counterpart (Button.Click) is observed at root and also when an ancestor already marked it handled, as IronPython's element-level handler always runs. The callback then runs after the ancestor's handlers instead of before them, which matters only if it depends on their side effects.

    Type Description
    PyRevitException

    The event is unknown, or is a direct routed event such as Loaded or MouseEnter that never reaches an ancestor, so it cannot be observed for template instances.

    def bind_template_event(registry, root, tag, uid, event_name, handler):
    """Route an event declared inside a template to ``handler``.

    Template instances do not exist when the XAML loads, so their elements are
    matched by ``Uid`` (and by type, for WPF controls). Bubbling events with a
    preview counterpart get a direct handler on each matching element the
    first time the tunnelling preview event reaches it, which preserves WPF's
    handler order and ``Handled`` semantics. Other routed events are matched
    from a handler on ``root``. Direct mouse-button events are rebuilt from
    ``MouseDown``/``MouseUp`` filtered by button and click count; double
    clicks accept the left or right button with ``ClickCount == 2``, as
    ``Control.MouseDoubleClick`` does, and a double click handler also runs
    after the press was marked handled, as that control's own event does. An element that unloads drops its
    direct handler and is rediscovered if it loads again. ``tag`` is
    the element's WPF type name, or ``{namespace}Name`` for a type outside the
    WPF namespace, which is then matched by ``Uid`` alone.

    Note:
        An event with no preview counterpart (``Button.Click``) is observed at
        ``root`` and also when an ancestor already marked it handled, as
        IronPython's element-level handler always runs. The callback then runs
        after the ancestor's handlers instead of before them, which matters
        only if it depends on their side effects.

    Raises:
        PyRevitException: The event is unknown, or is a direct routed event
            such as ``Loaded`` or ``MouseEnter`` that never reaches an
            ancestor, so it cannot be observed for template instances.
    """
    owner = _control_type(tag)
    mapped = _MOUSE_BUTTON_EVENTS.get(event_name)
    routed_name, button, clicks = mapped if mapped else (event_name, None, None)
    event = _routed_event(owner, routed_name)
    if event is None:
        raise PyRevitException(
            "Unsupported template event: {}.{}".format(tag, event_name)
        )
    if event.RoutingStrategy == Windows.RoutingStrategy.Direct:
        raise PyRevitException(
            "{}.{} is a direct routed event; CPython forms cannot observe it "
            "inside templates.".format(tag, event_name)
        )
    callback = registry.guard(handler)
    if mapped:
        callback = _mouse_button_filter(callback, button, clicks)

    def sources(args):
        return [
            element
            for element in _routing_ancestors(args.OriginalSource)
            if getattr(element, "Uid", None) == uid
            and (owner is None or isinstance(element, owner))
        ]

    preview = _routed_event(owner, "Preview" + routed_name)
    if event.RoutingStrategy == Windows.RoutingStrategy.Bubble and preview:
        _attach_on_preview(
            registry, root, event, preview, sources, callback, clicks == 2
        )
        return
    _attach_at_root(registry, root, event, sources, callback)

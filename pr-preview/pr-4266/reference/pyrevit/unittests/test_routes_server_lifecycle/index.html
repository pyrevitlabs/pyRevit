    test_routes_server_lifecycle

    Tests for the routes server lifecycle across a session reload.

    A reload tears the session down and builds a new one on a worker thread, so every step exercised here runs off Revit's STA UI thread, where an escaping exception terminates the host. Two properties matter:

    • stopping must not raise, and must release the port whatever else fails - a port left bound is a port the reloaded session cannot rebind;
    • the registration must be cleared even when the stop failed, otherwise the next activation hands the dead server back instead of binding, and the reloaded session silently serves no routes at all.

    Both were lost in the reload that killed Revit (#3473). Nothing on this path may write to stderr either, for the reason the request path may not: an exception printed off a worker thread is what terminated the process.

    def test_stop_fails_all_pending_requests_and_rejects_new_work(self):
    """Session reload releases every API waiter with a service error."""
    instance = self._make_routes_server()
    dispatcher = instance.server.request_handler
    called = []

    def route(request, uiapp):
        called.append(request.path)

    items = [
        handler.RequestWorkItem(base.Request(path="/%s" % index), route)
        for index in range(10)
    ]
    for item in items:
        self.assertIsNone(dispatcher.enqueue(item))
    instance.stop()
    dispatcher.Execute(None)
    rejected = handler.RequestWorkItem(base.Request(), route)
    self.assertIsNotNone(dispatcher.enqueue(rejected))
    items.append(rejected)

    for item in items:
        self.assertTrue(item.event.wait(0.1))
        self.assertEqual(503, item.response.status)
        parsed = handler.RequestHandler.parse_response(item.response)
        self.assertIn("stopped or reloaded", parsed.data)
        self.assertIn('"exception"', parsed.data)
    self.assertEqual([], called)

    def test_stop_notifies_queued_callbacks_outside_queue_lock(self):
    """Accepted callback requests receive their stopped-server rejection."""
    instance = self._make_routes_server()
    dispatcher = instance.server.request_handler
    callbacks = []
    notified = threading.Event()

    def callback(url, response):
        acquired = dispatcher._lock.acquire(False)
        if acquired:
            dispatcher._lock.release()
        callbacks.append((url, response.status, acquired))
        if len(callbacks) == 2:
            notified.set()
        else:
            raise ValueError("first callback failed")

    dispatcher.make_callback = callback
    for url in ("http://callback/first", "http://callback/second"):
        request = base.Request(data={"callbackUrl": url})
        item = handler.RequestWorkItem(request, lambda request, uiapp: None)
        self.assertIsNone(dispatcher.enqueue(item))

    instance.stop()
    self.assertTrue(notified.wait(5))
    instance.stop()
    self.assertEqual(
        [
            ("http://callback/first", 503, True),
            ("http://callback/second", 503, True),
        ],
        callbacks,
    )

    def test_handler_that_stops_server_keeps_its_response(self):
    """Reload initiated by an active route preserves sync and callback results."""
    for callback_url in (None, "http://callback/"):
        instance = self._make_routes_server()
        dispatcher = instance.server.request_handler
        callbacks = []
        uiapp = _StubUiApplication()
        external_event = instance.server.external_event

        def route(request, uiapp):
            instance.stop()
            self.assertEqual(0, external_event.dispose_calls)
            return {"session_id": "reloaded-session"}

        def callback(url, response):
            callbacks.append((url, response))
            self.assertEqual(0, external_event.dispose_calls)

        dispatcher.make_callback = callback
        request = base.Request(data={"callbackUrl": callback_url})
        item = handler.RequestWorkItem(request, route)
        queued = handler.RequestWorkItem(base.Request(), route)
        self.assertIsNone(dispatcher.enqueue(item))
        self.assertIsNone(dispatcher.enqueue(queued))

    def test_stop_gives_up_on_a_loop_that_never_acknowledges(self):
    """A shutdown that never returns must not hold the reload.

    BaseServer.shutdown blocks until the accept loop acknowledges, with no
    timeout of its own, so a wedged loop would keep the port bound and the
    registration alive - the reload would never finish. The wait is bounded
    instead, and the port is released on the way out either way.
    """
    http_server = _UnacknowledgedHttpServer()
    instance = self._make_routes_server(http_server=http_server)

    try:
        started = time.time()
        instance.stop(timeout=0.05)
        elapsed = time.time() - started

        self.assertLess(
            elapsed,
            _UnacknowledgedHttpServer.release_after / 2,
            "the stop waited for the shutdown instead of bounding it",
        )
        self.assertEqual(1, http_server.close_calls)
        self.assertIsNone(instance.server_thread)
        self.assertTrue(
            self.logger.has_errors(),
            "a loop that outlives the wait is reported, not silently dropped",
        )
    finally:
        http_server.release.set()

    def test_stop_spends_one_budget_across_both_waits(self):
    """A wedged loop must not cost the reload one timeout per wait.

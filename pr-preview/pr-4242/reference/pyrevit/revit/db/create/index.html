    def create_revision(
    description=None, by=None, to=None, date=None, alphanum=False, nonum=False, doc=None
):
    """Create a revision.

    Note:
        On Revit 2023+, numbering comes from a RevisionNumberingSequence, so
        ``alphanum`` assigns the document's first alphanumeric sequence and
        ``nonum`` clears the sequence.

    Raises:
        PyRevitException: ``alphanum`` is set on Revit 2023+ and the document
            has no alphanumeric numbering sequence.
    """
    doc = doc or DOCS.doc
    number_type = None
    if alphanum:
        number_type = DB.RevisionNumberType.Alphanumeric
    if nonum:
        number_type = coreutils.get_enum_none(DB.RevisionNumberType)
    sequence_id = (
        _find_revision_numbering_sequence(doc, number_type)
        if number_type is not None
        else None
    )
    new_rev = DB.Revision.Create(doc)
    new_rev.Description = description
    new_rev.IssuedBy = by or ""
    new_rev.IssuedTo = to or ""
    if number_type is not None:
        _set_revision_number_type(new_rev, number_type, sequence_id)
    new_rev.RevisionDate = date or ""
    return new_rev

    def copy_revisions(revisions, src_doc, dest_doc):
    if revisions is None:
        all_src_revs = query.get_revisions(doc=src_doc)
    else:
        all_src_revs = revisions

    def create_sheet(
    sheet_num, sheet_name, titleblock_id=DB.ElementId.InvalidElementId, doc=None
):
    """Create a sheet with a number and a name.

    Must run inside a transaction. Raises if the sheet number is already used.

    Args:
        sheet_num (str): sheet number, unique in the document.
        sheet_name (str): sheet name.
        titleblock_id (DB.ElementId): title block type id; the default
            creates a sheet without a title block.
        doc (DB.Document): target document; defaults to the active one.

    Returns:
        DB.ViewSheet: the new sheet.
    """
    doc = doc or DOCS.doc
    mlogger.debug("Creating sheet: %s - %s", sheet_num, sheet_name)
    mlogger.debug("Titleblock id is: %s", titleblock_id)
    newsheet = DB.ViewSheet.Create(doc, titleblock_id)
    newsheet.Name = sheet_name
    newsheet.SheetNumber = sheet_num
    return newsheet

    def create_3d_view(view_name, isometric=True, doc=None):
    doc = doc or DOCS.doc
    nview = query.get_view_by_name(view_name, doc=doc)
    if not nview:
        default_3dview_type = doc.GetDefaultElementTypeId(
            DB.ElementTypeGroup.ViewType3D
        )
        if isometric:
            nview = DB.View3D.CreateIsometric(doc, default_3dview_type)
        else:
            nview = DB.View3D.CreatePerspective(doc, default_3dview_type)

    nview.Name = view_name

    nview.CropBoxActive = False
    nview.CropBoxVisible = False
    if nview.CanToggleBetweenPerspectiveAndIsometric():
        if isometric:
            nview.ToggleToIsometric()
        else:
            nview.ToggleToPerspective()
    return nview

    def load_family_with_result(family_file, doc=None):
    """Load a family and return Revit's result with its available symbols.

    Args:
        family_file (str): Fully qualified path to the family file.
        doc (DB.Document): Target document. Defaults to the active document.

        tuple[bool, list[DB.FamilySymbol]]: The direct Revit load result and

        because a matching family is already present. Callers that report an
        overwrite must use the boolean result rather than symbol-list truthiness.

    # LoadFamily's out-param needs engine-specific marshaling: an explicit
    # clr.Reference under IronPython, a return tuple under pythonnet
    if IRONPY:
        ret_ref = clr.Reference[DB.Family]()
        loaded = doc.LoadFamily(family_file, FamilyLoaderOptionsHandler(), ret_ref)
        fam = ret_ref.Value
    else:
        loaded, fam = doc.LoadFamily(family_file, FamilyLoaderOptionsHandler(), None)

    if not loaded:
        # Family may already be loaded - check if the out-param has it
        if fam:
            mlogger.debug(
                "Family already loaded, retrieving symbols from document: %s",
                family_file,
            )
        else:
            # Try to find existing family by name from file path
            family_name = coreutils.get_file_name(family_file)
            mlogger.debug(
                "Family may already be loaded, attempting to retrieve by name: %s",
                family_name,
            )
            existing_families = query.get_family(family_name, doc=doc)
            if existing_families:
                # get_family returns FamilySymbol elements, which is what we need
                return False, list(existing_families)
            mlogger.debug(
                "Cannot load Family from file=%s and family not found in document.",
                family_file,
            )
            return False, fam_symbols

    for fam_symbol_id in fam.GetFamilySymbolIds():
        fam_symbol = doc.GetElement(fam_symbol_id)
        if fam_symbol:
            fam_symbols.append(fam_symbol)
    return bool(loaded), fam_symbols

    def create_param_value_filter(
    filter_name,
    param_id,
    param_values,
    evaluator,
    match_any=True,
    case_sensitive=False,
    exclude=False,
    category_list=None,
    doc=None,
):
    doc = doc or DOCS.doc

    rules = None
    param_prov = DB.ParameterValueProvider(param_id)

    # decide how to combine the rules
    logical_merge = DB.LogicalOrFilter if match_any else DB.LogicalAndFilter

    # create the rule set
    get_elementid_value = get_elementid_value_func()
    for pvalue in param_values:
        # grab the evaluator
        param_eval = PARAM_VALUE_EVALUATORS.get(evaluator, None)
        if not param_eval:
            raise PyRevitException("Unknown evaluator")

        # if value is str, eval is expected to be str
        str_eval, num_eval = param_eval
        if isinstance(pvalue, str):
            if HOST_APP.is_newer_than(2022):
                rule = DB.FilterStringRule(param_prov, str_eval(), pvalue)
            else:
                rule = DB.FilterStringRule(
                    param_prov, str_eval(), pvalue, case_sensitive
                )
        # if num_eval is for str, e.g. "contains", or "startswith"
        # convert numeric values to str
        elif isinstance(num_eval, DB.FilterStringRuleEvaluator):
            if isinstance(pvalue, (int, float)):
                if HOST_APP.is_newer_than(2022):
                    rule = DB.FilterStringRule(param_prov, num_eval(), str(pvalue))
                else:
                    rule = DB.FilterStringRule(
                        param_prov, num_eval(), str(pvalue), False
                    )
            elif isinstance(pvalue, DB.ElementId):
                p_id = str(get_elementid_value(pvalue))
                if HOST_APP.is_newer_than(2022):
                    rule = DB.FilterStringRule(param_prov, num_eval(), p_id)
                else:
                    rule = DB.FilterStringRule(param_prov, num_eval(), p_id, False)
        # if value is int, eval is expected to be numeric
        elif isinstance(pvalue, int):
            rule = DB.FilterIntegerRule(param_prov, num_eval(), pvalue)
        # if value is float, eval is expected to be numeric
        elif isinstance(pvalue, float):
            rule = DB.FilterDoubleRule(
                param_prov, num_eval(), pvalue, sys.float_info.epsilon
            )
        # if value is element id, eval is expected to be numeric
        elif isinstance(pvalue, DB.ElementId):
            rule = DB.FilterElementIdRule(param_prov, num_eval(), pvalue)
        if exclude:
            rule = DB.FilterInverseRule(rule)

        if rules:
            rules = logical_merge(rules, DB.ElementParameterFilter(rule))
        else:
            rules = DB.ElementParameterFilter(rule)

    # collect applicable categories
    if category_list:
        category_set = query.get_category_set(category_list, doc=doc)
    else:
        category_set = query.get_all_category_set(doc=doc)

    # filter the applicable categories
    filter_cats = []
    for cat in category_set:

            doc, framework.to_clr_list(DB.ElementId, [cat.Id]), rules
        ):
            filter_cats.append(cat.Id)

    # create filter
    return DB.ParameterFilterElement.Create(
        doc, filter_name, framework.to_clr_list(DB.ElementId, filter_cats), rules
    )

        so the family ends up twice as wide as the parameter. Label a
        two-plane dimension and add a second, unlabelled dimension through the
        centre plane with ``equal_segments=True``.

    if len(planes) < 2:
        raise PyRevitException("A dimension needs at least two reference planes.")
    if equal_segments and len(planes) < 3:
        raise PyRevitException("equal_segments needs at least three reference planes.")

            "Label a two-plane dimension; add a separate dimension with "
            "equal_segments=True for the centre plane."

    dimension = doc.FamilyCreate.NewDimension(view, line, references)
    if equal_segments:
        dimension.AreSegmentsEqual = True
    if label is not None:
        dimension.FamilyLabel = label
    return dimension

    def align_faces_to_planes(element, view, planes, tolerance=1e-6, doc=None):
    """Lock the planar faces of a family element to coincident reference planes.

    Every planar face that lies in one of ``planes`` gets an alignment
    constraint, so the element follows the planes when their dimensions change.

    Args:
        element (DB.Element): form such as an extrusion.
        view (DB.View): view the alignments are made in. A plan view reaches
            vertical faces, an elevation view reaches horizontal ones.
        planes (list[DB.ReferencePlane]): candidate planes.
        tolerance (float, optional): distance in feet within which a face counts
            as lying in a plane.
        doc (DB.Document, optional): family document, defaults to the active one.

    Returns:
        (list[str]): names of the planes that were aligned to a face.

    Note:
        A face that no plane contains is skipped. A point where two sloped
        faces meet has no face to align; give it a short flat edge longer than
        ``Application.ShortCurveTolerance``.
    """
    doc = doc or DOCS.doc
    _require_family_document(doc, "align faces")
    options = DB.Options()
    options.ComputeReferences = True
    options.View = view
    plane_geometry = [(plane, plane.GetPlane()) for plane in planes]
    aligned = []
    for geometry in element.get_Geometry(options):
        if not isinstance(geometry, DB.Solid):
            continue
        for face in geometry.Faces:
            if not isinstance(face, DB.PlanarFace):
                continue
            for plane, plane_def in plane_geometry:
                parallel = abs(abs(face.FaceNormal.DotProduct(plane_def.Normal)) - 1.0)
                distance = abs(
                    (face.Origin - plane_def.Origin).DotProduct(plane_def.Normal)
                )
                if parallel < tolerance and distance < tolerance:
                    doc.FamilyCreate.NewAlignment(
                        view, plane.GetReference(), face.Reference
                    )
                    aligned.append(plane.Name)
                    break
    return aligned

    def load_family_document(family_doc, doc=None, name=None):
    """Load an open family document into a project or another family.

    A family document that was never saved loads under its document title,
    such as ``Family24``; pass ``name`` to rename the loaded family.

    Args:
        family_doc (DB.Document): the family document to load.
        doc (DB.Document, optional): target document, defaults to the active one.
        name (str, optional): name for the loaded family.

    Returns:
        (DB.Family): the loaded family, in ``doc``.

    Raises:
        PyRevitException: when ``family_doc`` is not a family document, when
            ``doc`` has an open transaction, when another family in ``doc``
            already has the requested ``name``, or when ``doc`` already has a
            family named like ``family_doc`` and ``name`` would rename it. The
            names are checked before loading, so nothing is loaded in those
            cases.

    Important:
        Call it without an open transaction on ``doc``: ``LoadFamily`` manages
        its own. The rename runs in a transaction of its own, which is rolled
        back if Revit rejects the name.
        Reloading replaces the ``Family`` object, so a ``Family`` kept from an
        earlier load is no longer valid.

    Note:
        Revit matches an incoming family by its document title, not by the
        name it was renamed to. Loading the same unsaved document again does
        not update the renamed family; it adds a new family under the title.
        The name check refuses that case instead of leaving the duplicate.
    """
    doc = doc or DOCS.doc
    if not family_doc.IsFamilyDocument:
        raise PyRevitException("load_family_document needs a family document.")
    if doc.IsModifiable:
        raise PyRevitException(
            "Loading a family document must run outside a transaction on the target."
        )
    if name and name != family_doc.Title:
        taken = set(
            existing.Name
            for existing in DB.FilteredElementCollector(doc).OfClass(DB.Family)
        )
        if name in taken:
            raise PyRevitException(
                "A family named {!r} already exists in the target document.".format(
                    name
                )
            )
        if family_doc.Title in taken:
            raise PyRevitException(
                "Loading would overwrite the existing family {!r} and rename it. "
                "Load without a name to update that family.".format(family_doc.Title)
            )
    family = family_doc.LoadFamily(doc, FamilyLoaderOptionsHandler())
    if name and family.Name != name:
        rename = DB.Transaction(doc, "Rename loaded family")
        rename.Start()
        try:
            family.Name = name
            status = rename.Commit()
            if status != DB.TransactionStatus.Committed:
                raise PyRevitException(
                    "Revit did not apply the name {!r} to the loaded family "
                    "(transaction {}).".format(name, status)
                )
        except Exception:
            if rename.GetStatus() == DB.TransactionStatus.Started:
                rename.RollBack()
            raise
    return family

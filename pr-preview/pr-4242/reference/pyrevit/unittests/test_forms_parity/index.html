    test_forms_parity

    Interactive form contracts exercised identically on IronPython and CPython.

    These tests drive real WPF events with a bounded DispatcherTimer. They neither write model data nor require a person to answer the modal dialogs.

    IronPython is the reference: every test asserts the result IronPython's wpf.LoadComponent windows produce, and CPython must match it. Where the engines are known to differ by design, the test asserts both sides explicitly.

    def test_return_all_preserves_wrappers_and_reset(self):
    """Return-all and reset callbacks expose original TemplateListItem instances."""
    items = [
        forms.TemplateListItem("First"),
        forms.TemplateListItem("Second", checked=True),
    ]
    observed = []

    def reset(values):
        observed.extend(values)
        for value in values:
            value.checked = False

    def accept(dialog):
        _click(dialog.reset_b)
        _click(dialog.select_b)

    value = _modal(
        forms.SelectFromList,
        lambda: forms.SelectFromList.show(
            items, multiselect=True, return_all=True, resetfunc=reset
        ),
        accept,
    )
    self.assertEqual(items, observed)
    self.assertEqual(items, value)
    self.assertFalse(any(bool(item) for item in value))

    timer.Interval = framework.System.TimeSpan.FromMilliseconds(30)

        if not native.IsVisible:
            return
        timer.Stop()
        try:
            _click(window.pick_b)
        except Exception as error:
            failures.append(error)

    try:

            window.ShowDialog()
        _click(window.pick_b)
    finally:

        timer.Tick -= tick
        window.Close()
    if failures:
        raise failures[0]
    self.assertEqual(["pick", "pick", "pick"], window.seen)
